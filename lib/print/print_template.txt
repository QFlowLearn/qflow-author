#let data = json(bytes(sys.inputs.at("document")))
#let primary(language) = lower(language.split("-").first())
#set document(title: data.title, author: "QFlow", date: none)
#set text(font: "New Computer Modern", size: 11pt, lang: primary(data.locale))
#set par(leading: 0.65em, justify: false)
#show math.equation: set text(font: "New Computer Modern Math")
#let running-question = context {
  let page-number = counter(page).get().first()
  let markers = query(<qflow-question>)
  let previous = markers.filter(m => m.location().position().page < page-number)
  let current = markers.filter(m => m.location().position().page == page-number)
  if previous.len() > 0 and (current.len() == 0 or current.first().location().position().y > 30mm) {
    text(size: 8pt, data.labels.question + " " + str(previous.last().value) + " · " + data.labels.continued)
  }
}
#set page(paper: data.paper, margin: (x: 20mm, y: 20mm), header: align(right, running-question), footer: context align(right, text(size: 9pt)[#data.form · #counter(page).display("1")]))
#let render-math(node) = {
  if node.kind == "identifier" { math.italic(text(node.value)) }
  else if node.kind == "number" { text(node.value) }
  else if node.kind == "operator" { math.op(text(node.value), limits: false) }
  else if node.kind == "row" { node.children.map(render-math).join() }
  else if node.kind == "sup" { math.attach(render-math(node.left), t: render-math(node.right)) }
  else if node.kind == "sub" { math.attach(render-math(node.left), b: render-math(node.right)) }
  else if node.kind == "fraction" { math.frac(render-math(node.left), render-math(node.right)) }
  else if node.kind == "sqrt" { math.sqrt(render-math(node.body)) }
}
#let render-nodes(nodes, question: none) = nodes.map(node => {
  if node.kind == "text" { text(node.text) }
  else if node.kind == "break" { linebreak() }
  else if node.kind == "response" {
    render-nodes(node.prompt, question: question)
    if question.selection != "" { block(sticky: true, text(size: 9pt, question.selection)) }
    for choice in question.choices {
      block(breakable: false)[#text(choice.label + ". ")#render-nodes(choice.content)]
    }
    if data.kind == "student" and question.responseLines > 0 {
      block(breakable: false)[
        #text(size: 9pt, data.labels.question + " " + str(question.number) + " · " + data.labels.response)
        #for _ in range(question.responseLines) { v(6mm); line(length: 100%, stroke: 0.3pt + luma(65%)) }
      ]
    }
  }
  else {
    let body = if node.kind == "math" { math.equation(block: node.block, render-math(node.expression)) }
    else if node.kind == "image" {
      let asset = data.images.at(node.asset)
      block(breakable: false, image(asset, width: 100%, height: 60mm, fit: "contain", alt: node.alt))
    }
    else if node.kind == "paragraph" { par(render-nodes(node.children, question: question)) }
    else if node.kind == "strong" { strong(render-nodes(node.children, question: question)) }
    else if node.kind == "emphasis" { emph(render-nodes(node.children, question: question)) }
    else if node.kind == "list" { list(..node.children.map(child => render-nodes((child,), question: question))) }
    else if node.kind == "ordered-list" { enum(..node.children.map(child => render-nodes((child,), question: question))) }
    else if node.kind == "quote" { block(inset: (left: 5mm), render-nodes(node.children, question: question)) }
    else { render-nodes(node.children, question: question) }
    text(lang: primary(node.language), body)
  }
}).join()
#heading(level: 1, text(size: 24pt, data.title))
#text(size: 12pt, data.labels.document)
#v(4mm)
#text(size: 9pt, data.form)
#v(8mm)
#text(data.instructions)
#v(5mm)
#line(length: 100%, stroke: 0.5pt)
#for part in data.parts {
  heading(level: 2, text(size: 15pt, part.title))
  render-nodes(part.instructions)
  for entry in part.entries {
    if entry.kind == "content" { block(render-nodes(entry.content)); continue }
    [#metadata(entry.number) <qflow-question>]
    block(sticky: true, text(weight: "bold", data.labels.question + " " + str(entry.number)))
    render-nodes(entry.content, question: entry)
    if data.kind == "marking" {
      block(sticky: true, text(weight: "bold", data.labels.marking))
      for rule in entry.marking.rules { block(sticky: true, par(text(rule))) }
      render-nodes(entry.marking.guidance)
    }
    v(7mm)
  }
}
#context [#metadata(counter(page).final().first()) <qflow-page-count>]
